# unh-bench-scripts
Scripts for use with UNHBench, for automation/help

# CPU slowdown

MODE_KIND=slowdown_profile LATENCY_TARGETS_MS="25" MODES="cpu" CPU_STEPS="4 3 2 1" PROFILE_STEP_TARGETS="4=46000,3=42000,2=38000,1=34000" PROFILE_TRIALS=5 PROFILE_RUN_DURATION=60 ./runner.sh

# Network bandwidth slowdown

MODE_KIND=slowdown_profile LATENCY_TARGETS_MS="25" MODES="network_bandwidth" NETWORK_BANDWIDTH_STEPS="500 100 50 25 10" PROFILE_STEP_TARGETS="500=48000,100=25000,50=12000,25=5000,10=500" PROFILE_TRIALS=5 PROFILE_RUN_DURATION=60 ./runner.sh

# Network latency slowdown

MODE_KIND=slowdown_profile LATENCY_TARGETS_MS="25" MODES="network_latency" NETWORK_LATENCY_STEPS="0.25ms 0.5ms 1ms 5ms" PROFILE_STEP_TARGETS="0.25ms=48000,0.5ms=44000,1ms=42000,5ms=38000" PROFILE_TRIALS=5 PROFILE_RUN_DURATION=60 ./runner.sh

# All

MODE_KIND=slowdown_profile \
LATENCY_TARGETS_MS="25" \
MODES="cpu network_bandwidth network_latency" \
CPU_STEPS="4 3 2 1" \
NETWORK_BANDWIDTH_STEPS="500 100 50 25 10" \
NETWORK_LATENCY_STEPS="0.25ms 0.5ms 1ms 5ms" \
PROFILE_STEP_TARGETS="4=46000,3=42000,2=38000,1=34000,500=48000,100=25000,50=12000,25=5000,10=500,0.25ms=48000,0.5ms=44000,1ms=42000,5ms=38000" \
PROFILE_TRIALS=5 \
PROFILE_WARMUP_TRIALS=2 \
PROFILE_RUN_DURATION=60 \
./runner.sh
